# The string tables: XmStrDefs.c in this directory, and the headers in
# ${CMAKE_BINARY_DIR}/include/Xm, generated by makestrs from xmstring.list
# and the templates it names.
set(XM_STRING_TEMPLATES
    ${CMAKE_CURRENT_SOURCE_DIR}/XmStrDefs.ct
    ${CMAKE_CURRENT_SOURCE_DIR}/XmStrDefs.ht
    ${CMAKE_CURRENT_SOURCE_DIR}/XmStrDefs22.ht
    ${CMAKE_CURRENT_SOURCE_DIR}/XmStrDefs23.ht
    ${CMAKE_CURRENT_SOURCE_DIR}/XmStrDefsI.ht
)
set(XM_INCLUDE_FILES
    ${CMAKE_BINARY_DIR}/include/Xm/XmStrDefs.h
    ${CMAKE_BINARY_DIR}/include/Xm/XmStrDefs22.h
    ${CMAKE_BINARY_DIR}/include/Xm/XmStrDefs23.h
    ${CMAKE_BINARY_DIR}/include/Xm/XmStrDefsI.h
)
add_custom_command(
    OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/XmStrDefs.c ${XM_INCLUDE_FILES}
    COMMAND ${MOTIF_TOOL_ENV} ${CMAKE_COMMAND}
        -DMAKESTRS=$<TARGET_FILE:${MOTIF_HOST_TOOL_PREFIX}makestrs>
        -DSOURCE_DIR=${CMAKE_CURRENT_SOURCE_DIR}
        -DLIST=${CMAKE_CURRENT_SOURCE_DIR}/xmstring.list.in
        -DWORK_DIR=${CMAKE_CURRENT_BINARY_DIR}
        -DHEADER_DIR=${CMAKE_BINARY_DIR}/include/Xm
        -P ${CMAKE_SOURCE_DIR}/tools/cmake/scripts/generate_xm_strings.cmake
    DEPENDS
        ${MOTIF_HOST_TOOL_PREFIX}makestrs
        ${CMAKE_CURRENT_SOURCE_DIR}/xmstring.list.in
        ${XM_STRING_TEMPLATES}
        ${CMAKE_SOURCE_DIR}/tools/cmake/scripts/generate_xm_strings.cmake
    COMMENT "Generating Xm string definition files"
    VERBATIM
)

# Generate the message catalog header (XmMsgCatI.h) from the symbolic source
# catalog, or copy the stub when message catalogs are disabled
if(WITH_MESSAGE_CATALOG)
    motif_add_message_catalog(xm_msgcat_header Xm
        ${CMAKE_CURRENT_SOURCE_DIR}/Xm.msg
        ${CMAKE_BINARY_DIR}/include/Xm/XmMsgCatI.h
    )
else()
    add_custom_command(
        OUTPUT ${CMAKE_BINARY_DIR}/include/Xm/XmMsgCatI.h
        COMMAND ${CMAKE_COMMAND} -E make_directory ${CMAKE_BINARY_DIR}/include/Xm
        COMMAND ${CMAKE_COMMAND} -E copy
            ${CMAKE_SOURCE_DIR}/src/lib/Xm/XmMsgCatI.h
            ${CMAKE_BINARY_DIR}/include/Xm/XmMsgCatI.h
        COMMENT "Copying Xm message catalog stub"
    )
    add_custom_target(xm_msgcat_header ALL DEPENDS ${CMAKE_BINARY_DIR}/include/Xm/XmMsgCatI.h)
endif()

# Create a custom target to ensure string files are generated
add_custom_target(xm_strings ALL DEPENDS ${CMAKE_CURRENT_BINARY_DIR}/XmStrDefs.c ${XM_INCLUDE_FILES})

# Include directories
# Include directories.  In the build tree, the generated headers in
# ${CMAKE_BINARY_DIR}/include and the sources in src/lib come from
# motif_build_flags; src/lib/Xm is kept for in-tree users of the internal
# headers.
target_include_directories(Xm
    PUBLIC
        $<BUILD_INTERFACE:${CMAKE_SOURCE_DIR}/src/lib/Xm>
        $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>
    PRIVATE
        # ToolTipI.h includes "Xm.h", which is generated (from Xm.h.in)
        ${CMAKE_BINARY_DIR}/include/Xm
)

# Run-time search paths, compiled in as absolute paths.  The key bindings
# are installed by data/bindings, and the bitmaps by data/bitmaps under
# INCDIR.
target_compile_definitions(Xm
    PRIVATE
        XMBINDDIR_FALLBACK="${CMAKE_INSTALL_FULL_DATADIR}/X11/bindings"
        INCDIR="${CMAKE_INSTALL_FULL_INCLUDEDIR}/X11"
        LIBDIR="${CMAKE_INSTALL_FULL_LIBDIR}/X11"
        DATADIR="${CMAKE_INSTALL_FULL_DATADIR}"
)

# Link libraries.  X11 and Xt are part of the interface of libXm, and so is
# libXp with printing, as Xm.h then includes <X11/extensions/Print.h>.
set(XM_PUBLIC_LIBS PkgConfig::MOTIF_XT)
if(WITH_PRINTING)
    list(APPEND XM_PUBLIC_LIBS PkgConfig::MOTIF_XP)
endif()
set(XM_PRIVATE_LIBS PkgConfig::MOTIF_XEXT PkgConfig::MOTIF_FONTCONFIG)
if(MOTIF_XFT_FOUND)
    list(APPEND XM_PRIVATE_LIBS PkgConfig::MOTIF_XFT)
endif()
if(MOTIF_JPEG_FOUND)
    list(APPEND XM_PRIVATE_LIBS PkgConfig::MOTIF_JPEG)
endif()
if(MOTIF_PNG_FOUND)
    list(APPEND XM_PRIVATE_LIBS PkgConfig::MOTIF_PNG)
endif()
if(ICONV_LIBRARY)
    list(APPEND XM_PRIVATE_LIBS ${ICONV_LIBRARY})
endif()
target_link_libraries(Xm
    PUBLIC
        ${XM_PUBLIC_LIBS}
        $<BUILD_INTERFACE:motif_build_flags>
    PRIVATE
        ${XM_PRIVATE_LIBS}
        m
        Threads::Threads
)

# Add system libraries for missing functions
find_library(REGEX_LIB regex)
if(REGEX_LIB)
    target_link_libraries(Xm PRIVATE ${REGEX_LIB})
endif()

# Check for re_comp/re_exec library (some systems have this instead of regex)
find_library(REGEX_COMP_LIB regex)
if(NOT REGEX_COMP_LIB)
    find_library(REGEX_COMP_LIB re)
endif()
if(REGEX_COMP_LIB)
    target_link_libraries(Xm PRIVATE ${REGEX_COMP_LIB})
endif()

    RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR}
)
